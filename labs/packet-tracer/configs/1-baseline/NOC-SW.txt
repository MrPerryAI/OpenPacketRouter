! ============================================================
! NOC-SW  (Packet Tracer model: 3560-24PS)
! Role: FTI-lite core / Network Operations Center
!       Spanning-tree ROOT bridge for every VLAN
! Paste into the CLI tab, starting at the "Switch>" prompt.
! ============================================================
enable
configure terminal
hostname NOC-SW
no ip domain-lookup
banner motd # FTI-LITE CORE / NOC - Authorized access only #
!
vlan 100
 name SURVEILLANCE
vlan 200
 name VOICE
exit
!
spanning-tree mode rapid-pvst
spanning-tree vlan 1 priority 4096
spanning-tree vlan 100 priority 4096
spanning-tree vlan 200 priority 4096
!
interface FastEthernet0/1
 description TRUNK to ZFW-SW (Fort Worth Center)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
interface FastEthernet0/2
 description TRUNK to ZME-SW (Memphis Center)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
interface FastEthernet0/3
 description TRUNK to ZAB-SW (Albuquerque Center)
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
!
! Pre-provisioned for the El Dorado copper-to-fiber cutover (Part 2)
interface GigabitEthernet0/1
 description RESERVED - El Dorado new fiber uplink
 switchport trunk encapsulation dot1q
 switchport mode trunk
 no shutdown
!
interface FastEthernet0/10
 description NOC-MON workstation
 switchport mode access
 switchport access vlan 100
 spanning-tree portfast
 spanning-tree bpduguard enable
end
